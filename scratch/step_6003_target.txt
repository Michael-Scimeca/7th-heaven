                                          }).catch(err => console.error("Failed to sync resize:", err));

                                        });

                                      const handleMouseUp = () => {
                                        document.removeEventListener('mousemove', handleMouseMove);
                                        document.removeEventListener('mouseup', handleMouseUp);
                                      };
                                      
                                      document.addEventListener('mousemove', handleMouseMove);
                                      document.addEventListener('mouseup', handleMouseUp);
                                    }}
                                  >
                                    <div className="w-4 h-0.5 bg-white/25 rounded-full group-hover/resize:bg-white/60 transition-all" />
                                  </div>
                                </div>
                              );
                            })
                          })}

                                <span className="text-[8px] font-black text-amber-400 tracking-widest uppercase">Snapping To:</span>
                                <span className="text-[10px] font-bold text-white tracking-wide">
                                  {formatHour(dragGuide.startHour)}
                                </span>
                              </div>
                            </div>
                          )}
                          </div>

                          {/* Scroll Horizontal Reminder Overlay */}
                          {maxTotalCols > 2 && (
                            <div className="absolute bottom-2 inset-x-2 bg-black/95 border border-amber-500/20 text-amber-400 rounded px-2 py-1 text-[8px] font-black tracking-widest uppercase text-center select-none pointer-events-none z-30 shadow-[0_4px_10px_rgba(0,0,0,0.8)] flex items-center justify-center gap-1">
                              <span>↔ Scroll to view ({dayShifts.length})</span>
                            </div>
                          )}
                        </div>
                      );
                    })}
                  </div>
                </div>
              </>
            )}

            {calendarView === 'roster' && renderRosterBoard()}
            {calendarView === 'list' && renderListBoard()}
