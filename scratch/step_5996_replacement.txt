                    {/* 7 Days cells */}
                    {filteredDays.map(day => {
                      const dayShifts = schedules.filter(s => s.date === day.dateStr && s.crewId === member.id);
                      const dailyHours = dayShifts.reduce((sum, s) => sum + (s.endHour - s.startHour), 0);
                      const isOverworked = dailyHours > 8;

                      return (
                        <td 
                          key={day.dateStr} 
                          className={`p-2 border-r border-white/5 last:border-r-0 align-top transition-colors cursor-pointer relative ${
                            showCapacityHeatmap && isOverworked
                              ? 'bg-rose-500/10 hover:bg-rose-500/15'
                              : 'hover:bg-amber-500/[0.02]'
                          }`}
                          style={{ height: '240px' }}
                          onClick={(e) => {
                            e.stopPropagation();
                            setCellGroupPopover(`${member.id}_${day.dateStr}`);
                          }}
                          onDragOver={(e) => {
                            e.preventDefault();
                            if (e.dataTransfer) {
                              e.dataTransfer.dropEffect = "move";
                            }
                          }}
                          onDrop={(e) => {
                            e.preventDefault();
                            e.stopPropagation();
                            const dragData = e.dataTransfer.getData("text/plain");
                            if (dragData.startsWith("shift:")) {
                              const shiftId = dragData.split(":")[1];
                              
                              // Calculate vertical coordinates relative to the drop cell to determine new hours
                              const rect = e.currentTarget.getBoundingClientRect();
                              const y = e.clientY - rect.top;
                              const pct = Math.max(0, Math.min(1, y / rect.height));
                              const hoursFromStart = pct * 16;
                              // Snap to 30 mins
                              const snappedHours = Math.round(hoursFromStart * 2) / 2;
                              const dropHour = 8 + snappedHours;

                              setSchedules(current => {
                                const activeShift = current.find(s => s.id === shiftId);
                                if (!activeShift) return current;
                                
                                const duration = activeShift.endHour - activeShift.startHour;
                                const newStartHour = Math.max(8, Math.min(24 - duration, dropHour));
                                const newEndHour = newStartHour + duration;

                                const updated = current.map(s => {
                                  if (s.id === shiftId) {
                                    return {
                                      ...s,
                                      date: day.dateStr,
                                      crewId: member.id,
                                      crewName: member.id === 'openshifts' ? 'OpenShifts' : member.name,
                                      startHour: newStartHour,
                                      endHour: newEndHour,
                                      time: formatTimeFrame(newStartHour, newEndHour)
                                    };
                                  }
                                  return s;
                                });
                                localStorage.setItem('7h_crew_schedules', JSON.stringify(updated));
                                window.dispatchEvent(new Event('storage'));
                                // Sync to server
                                fetch("/api/crew/calendar", {
                                  method: "POST",
                                  headers: { "Content-Type": "application/json" },
                                  body: JSON.stringify(updated)
                                }).catch(err => console.error("Failed to sync schedules:", err));
                                return updated;
                              });
                            }
                          }}
                        >
                          {/* Cell group popover dropdown */}
                          {cellGroupPopover === `${member.id}_${day.dateStr}` && (
                            <div 
                              data-group-popover-cell 
                              className="absolute left-1/2 top-10 -translate-x-1/2 w-48 bg-[#111116] border border-white/10 rounded-xl shadow-2xl p-2.5 z-50 flex flex-col gap-1.5 animate-[scaleIn_0.15s_ease]"
                              onClick={(e) => e.stopPropagation()}
                            >
                              <div className="text-[9px] text-white/30 uppercase tracking-wider font-black px-1.5 pb-1 border-b border-white/5">
                                Actions: {day.dayName}
                              </div>
                              
                              {/* Option 1: Schedule this member */}
                              {member.id !== 'openshifts' && (
                                <button
                                  type="button"
                                  onClick={(e) => {
                                    e.stopPropagation();
                                    handleCellClick(day.dateStr, member.id, member.role || 'SERVER');
                                    setCellGroupPopover(null);
                                  }}
                                  className="w-full text-left px-2 py-1.5 rounded hover:bg-white/5 text-[10px] font-bold text-white transition-colors cursor-pointer border-none bg-transparent"
                                >
                                  👤 Schedule {member.name.split(' ')[0]}
                                </button>
                              )}

                              {/* Option 2: Add Open Shift */}
                              <button
                                type="button"
                                onClick={(e) => {
                                  e.stopPropagation();
                                  handleCellClick(day.dateStr, 'openshifts', member.role || 'SERVER');
                                  setCellGroupPopover(null);
                                }}
                                className="w-full text-left px-2 py-1.5 rounded hover:bg-white/5 text-[10px] font-bold text-emerald-400 transition-colors cursor-pointer border-none bg-transparent"
                              >
                                🟢 Add Open Shift
                              </button>

                              {/* Option 3: Apply Group / Create Group */}
                              <div className="border-t border-white/5 pt-1.5 flex flex-col gap-1">
                                <div className="text-[8px] text-white/20 uppercase tracking-widest font-black px-1.5 mb-0.5">
                                  Groups
                                </div>
                                
                                {crewGroups.map((g, idx) => (
                                  <button
                                    key={idx}
                                    type="button"
                                    onClick={(e) => {
                                      e.stopPropagation();
                                      handleAddGroupToDay(day.dateStr, g);
                                      setCellGroupPopover(null);
                                    }}
                                    className="w-full text-left px-2 py-1 rounded hover:bg-emerald-500/10 text-[9px] text-emerald-300 font-semibold transition-colors cursor-pointer border-none bg-transparent truncate"
                                    title={`Apply Group: ${g.name}`}
                                  >
                                    👥 {g.name}
                                  </button>
                                ))}

                                <button
                                  type="button"
                                  onClick={(e) => {
                                    e.stopPropagation();
                                    setCreateGroupForDate(day.dateStr);
                                    const initialSettings: any = {};
                                    crewMembers.filter(m => m.id !== 'openshifts').forEach(m => {
                                      initialSettings[m.id] = {
                                        active: false,
                                        role: m.role || 'SERVER',
                                        startHour: 17.0,
                                        endHour: 22.0
                                      };
                                    });
                                    setNewGroupMemberSettings(initialSettings);
                                    setNewGroupNameInput('');
                                    setIsCreateGroupModalOpen(true);
                                    setCellGroupPopover(null);
                                  }}
                                  className="w-full text-left px-2 py-1.5 rounded bg-emerald-500/10 hover:bg-emerald-500 hover:text-black text-[9px] font-black uppercase tracking-wider text-emerald-400 transition-all border border-emerald-500/20 hover:border-emerald-400 cursor-pointer"
                                >
                                  ➕ Create Group
                                </button>
                              </div>
                            </div>
                          )}

                          <div className="flex h-full relative select-none">
                            {/* Mini Time Axis (Left side of cell) */}
                            <div className="w-7 shrink-0 flex flex-col justify-between text-[9px] font-black text-white/30 pr-1.5 border-r border-white/5 select-none pointer-events-none py-2 leading-none">
                              <span>8a</span>
                              <span>12p</span>
                              <span>4p</span>
                              <span>8p</span>
                              <span>12a</span>
                            </div>
                            
                            {/* Inner absolute shift grid (Right side of cell) */}
                            <div className="flex-1 relative h-full bg-black/10 rounded overflow-hidden">
                              {dayShifts.map(shift => {
                                const topPct = ((shift.startHour - 8) / 16) * 100;
                                const heightPct = ((shift.endHour - shift.startHour) / 16) * 100;
                                return (
                                  <div
                                    key={shift.id}
                                    draggable
                                    onDragStart={(e) => {
                                      e.stopPropagation();
                                      e.dataTransfer.setData("text/plain", `shift:${shift.id}`);
                                      e.dataTransfer.effectAllowed = "move";
                                      setDraggedShiftId(shift.id);
                                    }}
                                    onDragEnd={() => setDraggedShiftId(null)}
                                    onClick={(e) => {
                                      e.stopPropagation();
                                      setEditingShiftId(shift.id);
                                      setDropStartHour(shift.startHour);
                                      setDropEndHour(shift.endHour);
                                      setDropRole(shift.role);
                                      setDropLocation(shift.location);
                                      setDropNotes(shift.notes || '');
                                    }}
                                    className="group/roster-card absolute inset-x-1 rounded-lg border bg-black/85 backdrop-blur-md p-1.5 cursor-grab active:cursor-grabbing hover:border-opacity-100 transition-all text-left flex flex-col justify-between overflow-hidden shadow-md"
                                    style={{
                                      top: `${topPct}%`,
                                      height: `${heightPct}%`,
                                      borderColor: crewColor
                                    }}
                                  >
                                    <div className="min-w-0 flex flex-col leading-tight">
                                      <div className="flex items-center justify-between gap-1">
                                        <span className="text-[10px] font-black text-white/95 truncate">
                                          {formatTimeFrame(shift.startHour, shift.endHour).split(' - ')[0]}
                                        </span>
                                      </div>
                                      <span className="text-[9px] font-bold text-white/50 truncate mt-0.5">{shift.role}</span>
                                    </div>
                                    <span className="text-[8px] text-amber-400/90 font-bold truncate mt-0.5">📍 {shift.location.split(',')[0]}</span>
                                    
                                    {/* Hover tooltip with full shift specs */}
                                    <div className="absolute left-full top-0 ml-2 w-44 p-2 bg-[#111116] border border-white/10 rounded shadow-xl z-50 pointer-events-none opacity-0 scale-95 group-hover/roster-card:opacity-100 group-hover/roster-card:scale-100 transition-all origin-left flex flex-col gap-0.5 backdrop-blur-md">
                                      <div className="text-[8px] font-black uppercase" style={{ color: crewColor }}>{shift.crewName}</div>
                                      <div className="text-[9px] font-bold text-white">{formatTimeFrame(shift.startHour, shift.endHour)}</div>
                                      <div className="text-[8px] text-white/40 uppercase tracking-wider font-semibold">{shift.role}</div>
                                      <div className="text-[8px] text-amber-400 font-bold truncate">📍 {shift.location}</div>
                                      {shift.notes && (
                                        <div className="text-[7px] text-white/70 italic mt-0.5 border-t border-white/5 pt-0.5 whitespace-pre-wrap">{shift.notes}</div>
                                      )}
                                    </div>
                                  </div>
                                );
                              })}
                            </div>
                          </div>
                          
                          {/* + ADD button */}
                          <button
                            type="button"
                            onClick={(e) => {
                              e.stopPropagation();
                              setCellGroupPopover(`${member.id}_${day.dateStr}`);
                            }}
                            className="mt-1 w-full py-1 flex items-center justify-center gap-1 text-[9px] font-bold text-white/25 hover:text-white/60 bg-transparent hover:bg-white/[0.03] border border-dashed border-white/15 hover:border-white/35 rounded transition-all cursor-pointer"
                          >
                            <svg width="8" height="8" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="3"><line x1="12" y1="5" x2="12" y2="19"/><line x1="5" y1="12" x2="19" y2="12"/></svg>
                            ADD
                          </button>
                        </td>
